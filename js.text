let result3 = prices.reduce(function(acc,current){
return acc+current;
}, 0)
console.log(result3)

// ==================================================
// المسألة الأولى - Map
//
// عندك Array من الأرقام.
// اعمل Array جديدة بحيث تضاعف كل رقم.
// ==================================================

let numbers1 = [3, 6, 9, 12];

let result = numbers1.map(function(element){
return element+element;
})
console.log(result)

// ==================================================
// المسألة الثانية - Filter
//
// عندك Array من الأرقام.
// احتفظ فقط بالأرقام الأكبر من 10.
// ==================================================

let numbers2 = [5, 15, 8, 20, 3, 17, 11];
let result2 = numbers2.filter(function(element){
return element > 10
})
console.log(result2)

// ==================================================
// المسألة الثالثة - Reduce
//
// عندك Array فيها أسعار منتجات.
// احسب إجمالي الأسعار كلها.
// ==================================================

let prices = [35, 50, 25, 80, 40];
let result3 = prices.reduce(function(acc,current){
return acc+current;
}, 0)
console.log(result3)


// ==================================================
// المسألة الرابعة - Map + Filter + Reduce
//
// عندك Array من الأرقام.
//
// 1 - احتفظ فقط بالأرقام الزوجية.
// 2 - اضرب كل رقم متبقي في 3.
// 3 - اجمع كل النتائج في قيمة واحدة.
//
// استخدم التلاتة:
// filter + map + reduce
// ==================================================

let numbers3 = [2, 5, 8, 10, 13, 16, 20];

let result4 = numbers3.filter(function(element){
return element % 2 === 0;
})
console.log(result4)


let result5 = result4.map(function(element){
return element*3
})
console.log(result5)




let result6 = result5.reduce(function(acc,current){
return acc+current;
}, 0)
console.log(result6)
